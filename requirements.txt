# Heimdall backend runtime dependencies (Python 3.11)
# Versions are intentionally not pinned yet. After a successful install and passing tests, run:
#   pip freeze > requirements.lock.txt
# and use the lock file to reproduce the exact environment.
 
# --- Web API ---
fastapi
uvicorn[standard]
python-multipart        # file uploads
pydantic-settings       # settings from environment variables / .env
 
# --- Database ---
sqlalchemy[asyncio]
asyncpg                 # async PostgreSQL driver
alembic                 # migrations
pgvector                # pgvector support for SQLAlchemy
 
# --- Background jobs and Redis ---
arq                     # job queue (installs the redis client with it)
 
# --- Security ---
argon2-cffi             # password hashing (argon2id)
 
# --- RAG: parsing, embeddings, reranking ---
pymupdf                 # PDF text extraction (AGPL-licensed; pypdf is a BSD-licensed alternative)
sentence-transformers   # embeddings and cross-encoder reranker (installs PyTorch)
 
# --- HTTP client (Ollama / OpenAI-compatible LLM calls) ---
httpx
